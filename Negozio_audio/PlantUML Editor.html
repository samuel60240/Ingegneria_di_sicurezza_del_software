' --- PRODOTTO E AZIONI ---
interface Supporto <<Interface>> {
    + $riproduzione():String
    + $registrazione(String):void

    + getTipo() : String
}

class Tape implements Supporto {
    - contenuto : String
    + scriviTraccia(traccia : String) : void
    + leggiTraccia() : String
    + getTipo() : String
}

class CD implements Supporto {
    - contenuto : String
    + scriviTraccia(traccia : String) : void
    + leggiTraccia() : String
    + getTipo() : String
}

' --- FACTORY METHOD ---
interface HiFiFactory <<Interface>> {
    + creaSupporto() : Supporto
    + masterizzatore(s : Supporto, traccia : String) : void
    + riproduttore(s : Supporto) : void
}

class TapeFactory implements HiFiFactory {

    + creaSupporto() : Tape

}

class CDFactory implements HiFiFactory {
    + creaSupporto() : CD

}

class CDProva<<singleton>> implements Supporto{
    - istanza : CDProva
    -CDProva()
    -getIstanza():CDProva
}

class TapeProva<<singleton>> implements Supporto{
    - istanza:TapeProva
    - TapeProva()
    + getIstanza():TapeProva
    

}


' --- SINGLETON (BANCO PROVA) ---
actor Cliente 

' --- RELAZIONI ---
TapeFactory ..> Tape : <<create>>
CDFactory ..> CD : <<create>>

Cliente ..>> HiFiFactory : compra

Cliente .> CDProva : prova
Cliente .> TapeProva :prova
